\begin{frame}[fragile]{Weisfeiler--Lehman hashing for tiling graphs}
\framesubtitle{1-WL colour refinement on the slot-level graph}

\textbf{Graph representation.}  Each slot $i \in \{0,\dots,n-1\}$ is a node.
Four \emph{typed} edges leave every node:
\[
  i \xrightarrow{\;\mathtt{R}\;} \mathtt{rneig}[i],\qquad
  i \xrightarrow{\;\mathtt{L}\;} \mathtt{lneig}[i],\qquad
  i \xrightarrow{\;\mathtt{M}\;} \mathtt{mirro}[i],\qquad
  i \xrightarrow{\;\mathtt{G}\;} \mathtt{glue}[i]
\]
Initial colour of node $i$: $\ell_i^{(0)} = \mathtt{polygon\_size}[i]$.

\vspace{2pt}
\textbf{Iterative refinement (3 rounds).}  Each round replaces every node's
colour by hashing its current colour together with the sorted multiset of its
neighbours' colours, \emph{prefixed by edge type} to distinguish $G$ from $M$:

\vspace{-2pt}
\[
  \ell_i^{(t+1)} = \operatorname{FNV}\!\Big(
    \ell_i^{(t)}\;,\;
    \textstyle\bigcup_{\tau\in\{R,L,M,G\}}\!\big\{\,
      \operatorname{FNV}(\text{seed}_\tau,\; \ell_{\tau(i)}^{(t)})
    \,\big\}
  \Big)_{\uparrow\text{sort}}
\]
\vspace{-2pt}

\textbf{FNV-1a hash} (64-bit, no MD5 dependency):
$\operatorname{FNV}(h, x) = (h \oplus x) \cdot 1099511628211$

\vspace{2pt}
\textbf{Final canonical string.}  Sort the $n$ refined colours and serialise
as raw bytes.  Identical tilings $\Rightarrow$ identical strings (invariant).

\vspace{2pt}
\footnotesize
\textbf{Properties:}  1-WL is \emph{necessary but not sufficient} for graph
isomorphism.  False positives are \emph{possible} (never observed on tiling
graphs) --- the full \texttt{solutions\_match} alias-refinement check serves as
a correctness fallback on hash collisions.  For tilings with all-uniform
polygon sizes, \textbf{2-WL} (pair-level) is available as \texttt{--wl-dim 2}.
\end{frame}
